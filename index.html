const KNOWN_SONGS = { "b0000.tbm": "Audition - BGM Intro Default", "b0001.tbm": "Audition - Let's Dance" }; const audio = document.getElementById('audio-player'); let playlist = []; let currentIndex = 0; function formatTime(seconds) { if (isNaN(seconds)) return "0:00"; const mins = Math.floor(seconds / 60); const secs = Math.floor(seconds % 60); return `${mins}:${secs < 10 ? '0' : ''}${secs}`; } // 1. Scraping Playlist Menggunakan Proxy Publik async function loadPlaylist() { try { const targetUrl = 'http://122.102.49.131/audition/ABM/'; const proxyUrl = 'https://corsproxy.io/?' + encodeURIComponent(targetUrl); const res = await fetch(proxyUrl); const html = await res.text(); const parser = new DOMParser(); const doc = parser.parseFromString(html, 'text/html'); doc.querySelectorAll('a').forEach(a => { const href = a.getAttribute('href'); if (href && href.endsWith('.tbm')) { const filename = href.split('/').pop(); playlist.push({ id: filename, title: KNOWN_SONGS[filename] || `Track (${filename})` }); } }); if(playlist.length > 0) { renderPlaylist(); document.getElementById('current-title').innerText = "Siap Diputar"; } } catch (err) { document.getElementById('current-title').innerText = "Gagal memuat server"; } } function renderPlaylist() { const container = document.getElementById('playlist-container'); container.innerHTML = ''; playlist.forEach((track, idx) => { const btn = document.createElement('button'); btn.className = `w-full text-left p-2.5 rounded-xl text-xs flex justify-between items-center transition ${idx === currentIndex ? 'bg-pink-500/20 text-pink-400 font-semibold' : 'text-slate-400 hover:bg-slate-800'}`; btn.innerHTML = `${track.title}${track.id}`; btn.onclick = () => selectTrack(idx); container.appendChild(btn); }); } // 2. Download dan Potong File async function selectTrack(index) { currentIndex = index; renderPlaylist(); const track = playlist[index]; document.getElementById('current-title').innerText = "Mengunduh Lagu (Harap Tunggu)..."; document.getElementById('current-file').innerText = track.id; try { const targetUrl = 'http://122.102.49.131/audition/ABM/' + track.id; const proxyUrl = 'https://corsproxy.io/?' + encodeURIComponent(targetUrl); const res = await fetch(proxyUrl); if (!res.ok) throw new Error("Gagal mengunduh"); const buffer = await res.arrayBuffer(); const view = new Uint8Array(buffer); let offset = -1; let mime = 'audio/mpeg'; // Cari header OGG atau MP3 for (let i = 0; i < view.length - 4; i++) { if (view[i] === 79 && view[i+1] === 103 && view[i+2] === 103 && view[i+3] === 83) { // OggS offset = i; mime = 'audio/ogg'; break; } if (view[i] === 73 && view[i+1] === 68 && view[i+2] === 51) { // ID3 offset = i; break; } if (view[i] === 255 && (view[i+1] === 251 || view[i+1] === 250 || view[i+1] === 243)) { // \xFF\xFB (MP3 Sync) offset = i; break; } } if (offset === -1) { document.getElementById('current-title').innerText = "Error: File Terenkripsi (XOR)"; return; } // Potong Header Audition const cleanBuffer = buffer.slice(offset); const blob = new Blob([cleanBuffer], { type: mime }); // Hapus URL memori lama agar tidak lemot if (audio.src) URL.revokeObjectURL(audio.src); const url = URL.createObjectURL(blob); audio.src = url; audio.play(); document.getElementById('current-title').innerText = track.title; document.getElementById('play-icon').classList.replace('fa-play', 'fa-pause'); } catch (err) { document.getElementById('current-title').innerText = "Koneksi ke server gagal"; } } // Controls document.getElementById('play-btn').onclick = () => { if (audio.paused) { audio.play(); document.getElementById('play-icon').classList.replace('fa-play', 'fa-pause'); } else { audio.pause(); document.getElementById('play-icon').classList.replace('fa-pause', 'fa-play'); } }; document.getElementById('next-btn').onclick = () => selectTrack((currentIndex + 1) % playlist.length); document.getElementById('prev-btn').onclick = () => selectTrack((currentIndex - 1 + playlist.length) % playlist.length); audio.ontimeupdate = () => { document.getElementById('seek-bar').value = (audio.currentTime / audio.duration) * 100 || 0; document.getElementById('current-time').innerText = formatTime(audio.currentTime); }; audio.onloadedmetadata = () => document.getElementById('duration').innerText = formatTime(audio.duration); audio.onended = () => document.getElementById('next-btn').click(); document.getElementById('seek-bar').oninput = (e) => audio.currentTime = (e.target.value / 100) * audio.duration; loadPlaylist();
